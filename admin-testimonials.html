⚙️ One-Time Firebase Setup Required
  1. Go to console.firebase.google.com → your project testimonial-36c85
  2. Enable Authentication: Build → Authentication → Sign-in method → Enable Email/Password
  3. Create admin user: Authentication → Users → Add user → Email: sandeepec035@gmail.com → set a strong password
  4. Update Firestore Rules: Firestore → Rules → paste the rules below → Publish
  5. Come back and sign in with sandeepec035@gmail.com and the password you set
// Paste this in Firestore → Rules → Publish
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /testimonials/{docId} {
      // Public: read approved only, create new
      allow read: if resource.data.approved == true;
      allow create: if true;
      // Admin: full access when signed in
      allow read, update, delete: if request.auth != null
        && request.auth.token.email == 'sandeepec035@gmail.com';
    }
  }
}
